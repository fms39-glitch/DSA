	public static String concat (Link head) {
		String result = "";
		Link c = head;
		while(c!=null) {
			result +=c.value;
			c = c.next;
		}
		return result;
	}

	public static int addNumbers (Link n1, Link n2) {
		Link a = n1;
		Link b = n2;
		int sum=0;
		int place=1;
		int carry = 0;
		// STUB
		while (a != null || b!=null || carry != 0) {
			int dig1 = (a != null) ? a.value:0;
			int dig2 = (b != null) ? b.value:0;
			int total = dig1+dig2+carry;
			
			carry = total/10;
			sum += (total%10)*place;
			place *= 10;
			
			if(a!=null) a=a.next;
			if(b!=null) b=b.next;
		}
		
		return sum;
	}

		public static Link merge(Link l1, Link l2) {
		    // Dummy node serves as a fixed starting anchor for the new list
		    Link dummy = new Link(0, null);
		    Link tail = dummy;

		    // Traverse both lists and attach the smaller node to tail
		    while (l1 != null && l2 != null) {
		        if (l1.value <= l2.value) {
		            tail.next = l1;
		            l1 = l1.next;
		        } else {
		            tail.next = l2;
		            l2 = l2.next;
		        }
		        tail = tail.next; // Move tail forward
		    }
		    if (l1 != null) {
		        tail.next = l1;
		    } else {
		        tail.next = l2;
		    }

		    // Skip the dummy node and return the actual merged head
		    return dummy.next;
		}

	public static Link reverseList(Link head) {
	    Link previous = null;
	    Link current = head;

	    while (current != null) {
	        Link next = current.next;
	        current.next = previous;
	        previous = current;
	        current = next;
	    }
	    return previous;
	}
	
	
	
	public static Link oddEvenList(Link head) {
	    if (head == null || head.next == null) {
	        return head; // List with 0 or 1 element is already sorted
	    }
	    // Dummy heads to build the two sub-lists easily
	    Link oddDummy = new Link(0, null);
	    Link evenDummy = new Link(0, null);
	    // Moving pointers for both sub-lists
	    Link oddTail = oddDummy;
	    Link evenTail = evenDummy;

	    Link curr = head;
	    while (curr != null) {
	        if (curr.value % 2 != 0) { 
	            // Odd value -> append to odd chain
	            oddTail.next = curr;
	            oddTail = oddTail.next;
	        } else {
	            // Even value -> append to even chain
	            evenTail.next = curr;
	            evenTail = evenTail.next;
	        }
	        curr = curr.next;
	    }

	    // Crucial: Terminate the even list to prevent cycle errors
	    evenTail.next = null;

	    // Connect the end of the odd list to the start of the even list
	    oddTail.next = evenDummy.next;

	    // Return the head of the combined list (skipping the odd dummy node)
	    return oddDummy.next;
	}
